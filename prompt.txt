```rust
use axum::middleware::{self, Next};
use axum::http::{Request, Response};
use uuid::Uuid;

async fn request_id_middleware<B>(
    mut req: Request<B>,
    next: Next<B>,
) -> Response<B> {
    let request_id = req
        .headers()
        .get("x-request-id")
        .and_then(|v| v.to_str().ok())
        .map(|s| s.to_string())
        .unwrap_or_else(|| Uuid::new_v4().to_string());

    req.extensions_mut().insert(request_id.clone());

    let mut response = next.run(req).await;

    response.headers_mut().insert(
        "x-request-id",
        request_id.parse().unwrap(),
    );

    response
}
```

在路由上添加 `.layer(middleware::from_fn(request_id_middleware))`。

```rust
use std::net::TcpListener;
use std::sync::Arc;
use mini_kv_core::WalEngine;
use mini_kv_client::MiniKvClient;
use mini_kv_server::{Args, run_server};

struct TestServer {
    base_url: String,
    shutdown_tx: tokio::sync::oneshot::Sender<()>,
}

impl TestServer {
    async fn start() -> Self {
        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
        let port = listener.local_addr().unwrap().port();
        let data_dir = tempfile::tempdir().unwrap();
        let data_dir_str = data_dir.path().to_string_lossy().to_string();

        let args = Args {
            data_dir: data_dir_str,
            host: "127.0.0.1".to_string(),
            port,
            log_level: "error".to_string(),
            wal_threshold_bytes: 67108864,
            janitor_interval_secs: 30,
        };

        let (shutdown_tx, shutdown_rx) = tokio::sync::oneshot::channel::<()>();

        tokio::spawn(async move {
            // 将 run_server 修改为支持 oneshot shutdown 信号
            // 或者直接在这里启动 server 并手动处理关闭
            tokio::select! {
                _ = run_server_with_args(args) => {}
                _ = shutdown_rx => {}
            }
        });

        // 等待 server 启动
        tokio::time::sleep(std::time::Duration::from_millis(200)).await;

        let base_url = format!("http://127.0.0.1:{}", port);
        TestServer { base_url, shutdown_tx }
    }

    fn client(&self) -> MiniKvClient {
        MiniKvClient::new(&self.base_url)
    }
}

impl Drop for TestServer {
    fn drop(&mut self) {
        let _ = self.shutdown_tx.send(());
    }
}
```

```markdown
# mini-kv

轻量级嵌入式 KV 存储引擎，纯 Rust 实现。

## 特性

- **WAL 持久化**：预写日志 + CRC32 校验，数据不丢失
- **TTL 过期**：支持毫秒级 key 过期，后台自动清理
- **前缀扫描**：利用 BTreeMap 有序性高效前缀匹配
- **HTTP API**：基于 axum 的 REST 接口
- **可观测性**：Prometheus 指标 + 结构化 JSON 日志 + X-Request-Id
- **优雅关闭**：SIGTERM/SIGINT 信号处理

## 项目结构

mini-kv/
├── Cargo.toml
├── crates/
│   ├── mini-kv-core/      # 核心存储引擎（WAL、内存索引、Janitor）
│   ├── mini-kv-server/     # HTTP API 服务（axum）
│   ├── mini-kv-client/     # Rust 客户端库
```
